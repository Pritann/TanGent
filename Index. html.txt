        // Database safety architecture setup
        if (!localStorage.getItem('users')) {
            localStorage.setItem('users', JSON.stringify({ teachers: [], students: [] }));
            localStorage.setItem('customQuestions', JSON.stringify([]));
            localStorage.setItem('studentGrades', JSON.stringify([]));

        // --- ANTI-INJECTION CHAR-FILTER SECURITY VARIABLE COMPLIANCE RULES ---
        document.querySelectorAll('.word-limit').forEach(input => {
            const counterLabel = input.nextElementSibling;
            const maxBound = parseInt(input.getAttribute('data-max')) || 8;
            const isEmail = input.classList.contains('email-field');
            const isName = input.classList.contains('name-field');
            
            function cleanInput(value) {
                let cleared = "";
                if (isEmail) {
                    cleared = value.replace(/[^a-zA-Z0-9@.]/g, '').toLowerCase();
                } else if (isName) {
                    cleared = value.replace(/[^a-zA-Z0-9 ]/g, '');
                } else {
                    cleared = value.replace(/[^a-zA-Z0-9]/g, '');
                }
                return cleared.substring(0, maxBound);
            }

            input.addEventListener('keydown', function(e) {
                const isControlKey = ['Backspace', 'Delete', 'ArrowLeft', 'ArrowRight', 'Tab', 'Enter'].includes(e.key);
                
                let validRegex = /^[a-zA-Z0-9]$/;
                if (isEmail) {
                    validRegex = /^[a-zA-Z0-9@.]$/;
                } else if (isName) {
                    validRegex = /^[a-zA-Z0-9 ]$/;
                }
                
                if (!isControlKey && this.value.length >= maxBound) {
                    e.preventDefault();
                    return;
                }
                if (!isControlKey && !validRegex.test(e.key)) {
                    e.preventDefault();
                }
            });

            input.addEventListener('input', function(e) {
                this.value = cleanInput(this.value);
                
                if (counterLabel && counterLabel.classList.contains('word-counter')) {
                    counterLabel.innerText = `${this.value.length} / ${maxBound} letters`;
                }
            });

            input.addEventListener('paste', function(e) {
                e.preventDefault();
                const pastedText = (e.clipboardData || window.clipboardData).getData('text');
                this.value = cleanInput(this.value + pastedText);
                
                if (counterLabel && counterLabel.classList.contains('word-counter')) {
                    counterLabel.innerText = `${this.value.length} / ${maxBound} letters`;
                }
            });
        });

        // --- DEFAULT STANDARDIZED ASSESSMENT Pools ---

            ],

        let activeExam = { questions: [], answers: {}, currentIdx: 0, timeLeft: 0, timer: null };

        function switchTab(tab) {
            document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
            document.getElementById(`tab-${tab}`).classList.add('active');
            document.getElementById('student-auth').classList.toggle('hidden', tab !== 'student');
            document.getElementById('teacher-auth').classList.toggle('hidden', tab !== 'teacher');
        }

        function toggleForm(type, action) {
            document.getElementById(`${type}-signin`).classList.toggle('active-form', action === 'signin');
            document.getElementById(`${type}-signup`).classList.toggle('active-form', action === 'signup');
        }

        function showView(viewId) {
            document.querySelectorAll('.view-section, .exam-overlay').forEach(el => el.classList.add('hidden'));
            document.getElementById(viewId).classList.remove('hidden');
            if (viewId === 'student-dashboard') {
                renderStudentExams();
            }
        }

        function logout() {
            activeUser = null;
            clearInterval(activeExam.timer);
            showView('auth-portal');
        }

            }

        document.getElementById('create-question-form').addEventListener('submit', function(e) {
            e.preventDefault();
            
            let opts = [
                document.getElementById('cq-a').value,
                document.getElementById('cq-b').value,
                document.getElementById('cq-c').value,
                document.getElementById('cq-d').value
            ];
            let optE = document.getElementById('cq-e').value;

            const newQ = {

                q: document.getElementById('cq-q').value,

            };

            let customPool = JSON.parse(localStorage.getItem('customQuestions')) || [];
            customPool.push(newQ);
            localStorage.setItem('customQuestions', JSON.stringify(customPool));

            this.reset();

        });

        function updateLiveQuestionList() {
            const customPool = JSON.parse(localStorage.getItem('customQuestions')) || [];
            const container = document.getElementById('live-question-list');
            const myQuestions = customPool.filter(q => q.authorCode === activeUser.code);
            
            if(myQuestions.length === 0) {
                container.innerHTML = `<p style="color: var(--text-secondary); text-align: center; padding: 20px;">No custom questions generated inside this sandbox session yet.</p>`;
                return;
            }

            container.innerHTML = "";
            myQuestions.forEach((q, idx) => {
                container.innerHTML += `
                    <div class="recorded-q-item">
                        <div class="recorded-q-header">
                            <span>Subject: ${q.subject} | Limit: ${q.duration} Mins</span>
                            <span style="color: var(--accent-glow)">Matrix Key: ${q.correct}</span>
                        </div>
                        <div style="font-weight: 600; margin-bottom: 8px;">Q${idx+1}: ${q.q}</div>
                        <div style="font-size: 0.85rem; color: var(--text-secondary)">
                            Options: ${q.options.map((o, i) => `[${String.fromCharCode(65+i)}] ${o}`).join(' | ')}
                        </div>
                    </div>
                `;
            });
        }

            tbody.innerHTML = "";
            grades.forEach(g => {
                tbody.innerHTML += `<tr><td>${g.studentName}</td><td>${g.exam}</td><td>${g.score}</td></tr>`;
            });
        }

        // --- RUNTIME SCHEDULER MATRIX ---
        function startExam(type) {
            let compiledSet = [];
            let examMinutes = 180; 
            
            if (type === 'Teacher Exam') {
                const customPool = JSON.parse(localStorage.getItem('customQuestions')) || [];
                const teacherQuestions = customPool.filter(q => q.authorCode === activeUser.teacherCode);
                
                if(teacherQuestions.length > 0) {
                    examMinutes = teacherQuestions[0].duration || 30;
                }

                teacherQuestions.forEach((tq, i) => {
                    compiledSet.push({
                        number: i + 1,
                        text: tq.q,
                        options: tq.options,
                        correct: tq.correct
                    });
                });
            } else {
                let pool = pyqPools[type];
                let numQuestions = 180;
                for(let i = 1; i <= numQuestions; i++) {
                    let template = pool[(i - 1) % pool.length];
                    compiledSet.push({
                        number: i,
                        text: template.q,
                        options: template.options,
                        correct: template.correct
                    });
                }
            }

            activeExam = {
                type: type,
                questions: compiledSet,
                answers: {},
                currentIdx: 0,
                timeLeft: examMinutes * 60
            };

            document.getElementById('active-exam-title').innerText = `${type} Workspace Profile`;
            showView('exam-engine');
            
            renderGrid();
            renderQuestion();

            clearInterval(activeExam.timer);
            activeExam.timer = setInterval(() => {
                if(activeExam.timeLeft <= 0) {
                    submitExam(true);
                    return;
                }
                activeExam.timeLeft--;
                
                let h = Math.floor(activeExam.timeLeft / 3600).toString().padStart(2, '0');
                let m = Math.floor((activeExam.timeLeft % 3600) / 60).toString().padStart(2, '0');
                let s = (activeExam.timeLeft % 60).toString().padStart(2, '0');
                document.getElementById('timer-display').innerText = `${h}:${m}:${s}`;
            }, 1000);
        }

        function renderGrid() {
            const grid = document.getElementById('q-grid-container');
            grid.innerHTML = "";
            activeExam.questions.forEach((q, idx) => {
                let cls = activeExam.currentIdx === idx ? "active" : "";
                if(activeExam.answers[idx]) cls = "answered";
                grid.innerHTML += `<button class="q-btn ${cls}" onclick="jumpQuestion(${idx})">${q.number}</button>`;
            });
        }

        function renderQuestion() {
            let q = activeExam.questions[activeExam.currentIdx];
            document.getElementById('q-number-display').innerText = `Question ${q.number} of ${activeExam.questions.length}`;
            document.getElementById('q-text-display').innerText = q.text;

            const opts = document.getElementById('options-container');
            opts.innerHTML = "";
            let letters = ["A", "B", "C", "D", "E"];
            
            q.options.forEach((optText, i) => {
                let l = letters[i];
                let checked = activeExam.answers[activeExam.currentIdx] === l ? "checked" : "";
                opts.innerHTML += `
                    <label class="option-label">
                        <input type="radio" name="exam-opt" value="${l}" ${checked} onchange="saveAnswer('${l}')">
                        <strong>${l}.</strong> ${optText}
                    </label>
                `;
            });
        }

        function saveAnswer(letter) {
            activeExam.answers[activeExam.currentIdx] = letter;
            renderGrid(); 
        }

        function navQuestion(dir) {
            let newIdx = activeExam.currentIdx + dir;
            if(newIdx >= 0 && newIdx < activeExam.questions.length) {
                jumpQuestion(newIdx);
            }
        }

        function jumpQuestion(idx) {
            activeExam.currentIdx = idx;
            renderQuestion();
            renderGrid();
        }

        function exitExam() {
            if(confirm("Confirm Early Termination? Unsaved items commit into records ledger.")) {
                submitExam(true);
            }
        }

        function submitExam(force = false) {
            if(!force && !confirm("Commit final choices ledger to database evaluation engine?")) return;
            
            clearInterval(activeExam.timer);
            let correct = 0;
            activeExam.questions.forEach((q, idx) => {
                if(activeExam.answers[idx] === q.correct) correct++;
            });

            let scoreText = `${correct} / ${activeExam.questions.length}`;
            let grades = JSON.parse(localStorage.getItem('studentGrades')) || [];
            grades.push({
                studentName: activeUser.name,
                exam: activeExam.type,
                score: scoreText
            });
            localStorage.setItem('studentGrades', JSON.stringify(grades));

            alert(`Assessment Vector Successfully Evaluated!\nResult Matrix: ${scoreText}`);
            showView('student-dashboard');
        }
